// === MATH UTILITIES ===
function boxMuller() {
  let u;
  do { u = Math.random(); } while (u === 0);
  const v = Math.random();
  return Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * v);
}
 
function normCDF(x) {
  const a1=0.254829592,a2=-0.284496736,a3=1.421413741,a4=-1.453152027,a5=1.061405429,p=0.3275911;
  const sign = x < 0 ? -1 : 1;
  const ax = Math.abs(x);
  const t = 1/(1+p*ax);
  const y = 1-(((((a5*t+a4)*t)+a3)*t+a2)*t+a1)*t*Math.exp(-ax*ax);
  return 0.5*(1+sign*y);
}
 
function bsCall(S, K, T, sigma) {
  if (T <= 0 || S <= 0) return Math.max(S - K, 0);
  const sqT = Math.sqrt(T);
  const d1 = (Math.log(S/K) + 0.5*sigma*sigma*T) / (sigma*sqT);
  const d2 = d1 - sigma*sqT;
  return S*normCDF(d1) - K*normCDF(d2);
}
 
function bsPut(S, K, T, sigma) {
  if (T <= 0 || S <= 0) return Math.max(K - S, 0);
  const sqT = Math.sqrt(T);
  const d1 = (Math.log(S/K) + 0.5*sigma*sigma*T) / (sigma*sqT);
  const d2 = d1 - sigma*sqT;
  return K*normCDF(-d2) - S*normCDF(-d1);
}

function generatePath() {

  path[0] = S0;
  for (let i = 1; i <= T21_STEPS; i++) {
    const z = boxMuller();
    path[i] = path[i-1] * Math.exp((-0.5*SIGMA*SIGMA)*DT + SIGMA*Math.sqrt(DT)*z);
  }
  return path;
}
 
// === PAYOFF COMPUTATION ===
function computePayoff(path, contract) {
  const ST = path[contract.expirySteps];
  switch(contract.type) {
    case 'Underlying': return ST;
    case 'Call': return Math.max(ST - contract.strike, 0);
    case 'Put':  return Math.max(contract.strike - ST, 0);
    case 'BinaryPut': return ST < contract.strike ? BINARY_NOTIONAL : 0;
    case 'KnockOutPut':
      for (let i = 0; i <= contract.expirySteps; i++) {
        if (path[i] < contract.barrier) return 0;
      }
      return Math.max(contract.strike - ST, 0);
    case 'Chooser': {
      const Sc = path[contract.chooseStep];
      const rem = (contract.expirySteps - contract.chooseStep) / STEPS_PER_YEAR;
      return Math.max(bsCall(Sc, contract.strike, rem, SIGMA),
                      bsPut(Sc,  contract.strike, rem, SIGMA));
    }
    default: return 0;
  }
}
 
function computePortfolioPnL(path, pos) {
  let total = 0;
  for (let i = 0; i < CONTRACTS.length; i++) {
    const p = pos[i];
    if (p.side === 'none' || p.volume === 0) continue;
    const c = CONTRACTS[i];
    const payoff = computePayoff(path, c);
    const entry = p.side === 'buy' ? c.ask : c.bid;
    if (p.side === 'buy') {
      total += (payoff - entry) * p.volume * CONTRACT_SIZE;
    } else {
      total += (entry - payoff) * p.volume * CONTRACT_SIZE;
    }
  }
  return total;
}
 
function computeContractPnL(path, contractIdx, pos) {
  const p = pos[contractIdx];
  if (p.side === 'none' || p.volume === 0) return 0;
  const c = CONTRACTS[contractIdx];

  return p.side === 'buy'
    ? (payoff - entry) * p.volume * CONTRACT_SIZE
    : (entry - payoff) * p.volume * CONTRACT_SIZE;
}

// === STATISTICS ===
function computeStats(pnls) {
  const n = pnls.length;
  let sum = 0;
  for (let i = 0; i < n; i++) sum += pnls[i];
  const mean = sum / n;
 
  let varSum = 0;
  for (let i = 0; i < n; i++) varSum += (pnls[i] - mean) ** 2;
  const std = Math.sqrt(varSum / n);
 
  const sorted = Float64Array.from(pnls).sort();
  const median = n % 2 === 0
    ? (sorted[n/2-1] + sorted[n/2]) / 2
    : sorted[Math.floor(n/2)];
  const p5  = sorted[Math.floor(n * 0.05)];
  const p95 = sorted[Math.floor(n * 0.95)];
 
  let profCount = 0;
  for (let i = 0; i < n; i++) if (pnls[i] > 0) profCount++;
 
  return { mean, std, median, p5, p95, sharpe: std > 0 ? mean/std : 0, profPct: profCount/n*100 };
}

  return v.toFixed(0);
}

// === PAYOFF DIAGRAM (analytical) ===
function getAnalyticalPayoff(S, contractIdx, pos) {
  const p = pos[contractIdx];
  if (p.side === 'none' || p.volume === 0) return 0;
  const c = CONTRACTS[contractIdx];
  let payoff;
  switch(c.type) {
    case 'Underlying': payoff = S; break;
    case 'Call': payoff = Math.max(S - c.strike, 0); break;
    case 'Put':  payoff = Math.max(c.strike - S, 0); break;
    case 'BinaryPut': payoff = S < c.strike ? BINARY_NOTIONAL : 0; break;
    case 'KnockOutPut': payoff = 0; break; // hard to show analytically — show as 0 (knocked)
    case 'Chooser': {
      const rem = (c.expirySteps - c.chooseStep) / STEPS_PER_YEAR;
      payoff = Math.max(bsCall(S, c.strike, rem, SIGMA), bsPut(S, c.strike, rem, SIGMA));
      break;
    }
    default: payoff = 0;
  }
  const entry = p.side === 'buy' ? c.ask : c.bid;
  if (c.type === 'Underlying') {
    return p.side === 'buy'
      ? (payoff - entry) * p.volume * CONTRACT_SIZE
      : (entry - payoff) * p.volume * CONTRACT_SIZE;
  }
  return p.side === 'buy'
    ? (payoff - entry) * p.volume * CONTRACT_SIZE
    : (entry - payoff) * p.volume * CONTRACT_SIZE;
}
 
function updatePayoffDiagram() {
  const canvas = document.getElementById('payoffCanvas');
  const ctx = canvas.getContext('2d');
  const W = canvas.offsetWidth || canvas.parentElement.offsetWidth;
  const H = 140;
  canvas.width = W;
  canvas.height = H;
  ctx.clearRect(0, 0, W, H);
 
  const prices = [];
  for (let s = 5; s <= 100; s += 1) prices.push(s);
  const payoffs = prices.map(S => {
    let total = 0;
    for (let i = 0; i < CONTRACTS.length; i++) total += getAnalyticalPayoff(S, i, positions);
    return total;
  });
 
  const minP = Math.min(...payoffs);
  const maxP = Math.max(...payoffs);
  const range = maxP - minP || 1;
  const pad = 20;
 
  const toX = (i) => pad + (i / (prices.length - 1)) * (W - 2*pad);
  const toY = (v) => (H - pad) - ((v - minP) / range) * (H - 2*pad);
 
  // Zero line
  const y0 = toY(0);
  ctx.strokeStyle = 'rgba(139,148,158,0.4)';
  ctx.lineWidth = 1;
  ctx.setLineDash([4,4]);
  ctx.beginPath();
  ctx.moveTo(pad, y0);
  ctx.lineTo(W - pad, y0);
  ctx.stroke();
  ctx.setLineDash([]);
 
  // Payoff curve
  ctx.strokeStyle = '#00ff88';
  ctx.lineWidth = 2;
  ctx.shadowColor = 'rgba(0,255,136,0.4)';
  ctx.shadowBlur = 6;
  ctx.beginPath();
  payoffs.forEach((v, i) => {
    const x = toX(i), y = toY(v);
    i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
  });
  ctx.stroke();
  ctx.shadowBlur = 0;
 
  // Fill
  ctx.beginPath();
  payoffs.forEach((v, i) => {
    const x = toX(i), y = toY(v);
    i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
  });
  ctx.lineTo(toX(payoffs.length-1), y0);
  ctx.lineTo(toX(0), y0);
  ctx.closePath();
  ctx.fillStyle = 'rgba(0,255,136,0.07)';
  ctx.fill();
 
  // Current price line
  const s0idx = prices.indexOf(50);
  if (s0idx >= 0) {
    const xs0 = toX(s0idx);
    ctx.strokeStyle = 'rgba(0,229,255,0.5)';
    ctx.lineWidth = 1;
    ctx.setLineDash([3,3]);
    ctx.beginPath();
    ctx.moveTo(xs0, pad);
    ctx.lineTo(xs0, H - pad);
    ctx.stroke();
    ctx.setLineDash([]);
    ctx.fillStyle = 'rgba(0,229,255,0.8)';
    ctx.font = '9px Share Tech Mono';
    ctx.fillText('S₀', xs0 + 3, pad + 10);
  }
 
  // Axes labels
  ctx.fillStyle = '#8b949e';
  ctx.font = '9px Share Tech Mono';
  ctx.fillText('5', pad, H - 4);
  ctx.fillText('100', W - pad - 20, H - 4);
  ctx.fillText(fmt(maxP), pad + 2, pad + 10);
  ctx.fillText(fmt(minP), pad + 2, H - pad - 4);
}

function renderTable() {
  const tbody = document.getElementById('tableBody');
  tbody.innerHTML = '';
  CONTRACTS.forEach((c, i) => {
    const pos = positions[i];
    const tr = document.createElement('tr');
    tr.id = `row_${i}`;
    tr.className = pos.side === 'buy' ? 'row-buy' : pos.side === 'sell' ? 'row-sell' : '';
 
    const expiryLabel = EXPIRY_LABEL[c.expirySteps] || `T + ${c.expirySteps/4}`;
    const typeClass = TYPE_CLASS[c.type] || 'type-vanilla';

    tr.innerHTML = `
      <td><span class="opt-name">${c.name}</span></td>
      <td>
        <div class="tooltip-wrap">
          <span class="type-badge ${typeClass}">${isExotic?'✦ ':''}${c.type}</span>
          <div class="tooltip-text">${tooltip}</div>

function setSide(i, val, selectEl) {
  positions[i].side = val;
  selectEl.className = `side-select side-${val}`;
  const row = document.getElementById(`row_${i}`);
  row.className = val === 'buy' ? 'row-buy' : val === 'sell' ? 'row-sell' : '';
  document.getElementById('presetSelect').value = 'custom';
  updatePayoffDiagram();
}
 
function setVolume(i, val) {
  let v = parseInt(val) || 0;
  const max = CONTRACTS[i].maxVol;
  if (v < 0) v = 0;
  if (v > max) v = max;
  positions[i].volume = v;
  document.getElementById('presetSelect').value = 'custom';
  updatePayoffDiagram();
}
